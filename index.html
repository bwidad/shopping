// URL API const API_URL = "https://mall-back-hxfbcrhuh7bnd7ge.westeurope-01.azurewebsites.net/api/malls"; // DATA dynamique let malls = []; // DOM const mallListEl=document.getElementById("mall-list"); const searchInput=document.getElementById("search"); const cityFilter=document.getElementById("cityFilter"); const showMalls=document.getElementById("showMalls"); const showMarjane=document.getElementById("showMarjane"); const detailPanel=document.getElementById("detail-panel"); const closeDetail=document.getElementById("close-detail"); const mallTitle=document.getElementById("mall-title"); const mallCity=document.getElementById("mall-city"); const mallAddress=document.getElementById("mall-address"); const mallCoords=document.getElementById("mall-coords"); const storesEl=document.getElementById("stores"); const fastfoodsEl=document.getElementById("fastfoods"); let map,markerCluster; // ================= FETCH API ================= async function loadMalls(){ try{ const response = await fetch(API_URL); const data = await response.json(); // Adapter structure pour ton app malls = data.map(m => ({ ...m, fastfood: [], // API ne fournit pas fastfood type: "mall" // tous sont des malls })); initCityFilter(); renderMalls(); }catch(error){ console.error("Erreur chargement API:", error); } } // ================= INIT FILTRES ================= function initCityFilter(){ cityFilter.innerHTML=''; [...new Set(malls.map(m=>m.city))].forEach(city=>{ const opt=document.createElement("option"); opt.value=city; opt.textContent=city; cityFilter.appendChild(opt); }); } // ================= RENDER LIST & MAP ================= function renderMalls(){ mallListEl.innerHTML=""; const search=searchInput.value.toLowerCase(); const selectedCity=cityFilter.value; const filtered=malls.filter(m=>m.name.toLowerCase().includes(search) && (selectedCity===""||m.city===selectedCity) ); filtered.forEach(m=>{ const card=document.createElement("div"); card.className="mall-card"; card.innerHTML=`

${m.name}

${m.city}

${m.stores.length} magasins

`; card.onclick=()=>showDetail(m); mallListEl.appendChild(card); }); renderMap(filtered); } // ================= RENDER MAP ================= function renderMap(filteredMalls){ if(map) map.remove(); map=L.map('map').setView([33.5731,-7.7076],5); L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png',{ attribution:'© OpenStreetMap' }).addTo(map); markerCluster=L.markerClusterGroup(); filteredMalls.forEach(m=>{ const icon=L.divIcon({ html:"🏬", className:"", iconSize:[30,30], iconAnchor:[15,15] }); const marker=L.marker( [m.location.lat,m.location.lng], {icon:icon} ).bindPopup(`${m.name}
${m.location.address}`); markerCluster.addLayer(marker); }); map.addLayer(markerCluster); if(cityFilter.value){ const cityMalls=filteredMalls.filter(m=>m.city===cityFilter.value); if(cityMalls.length){ const bounds=L.latLngBounds(cityMalls.map(m=>[ m.location.lat, m.location.lng ])); map.fitBounds(bounds,{padding:[50,50]}); } } } // ================= SHOW DETAIL ================= function showDetail(m){ detailPanel.style.display="block"; mallTitle.textContent=m.name; mallCity.textContent="Ville: "+m.city; mallAddress.textContent="Adresse: "+m.location.address; mallCoords.textContent=`Lat: ${m.location.lat}, Lng: ${m.location.lng}`; storesEl.innerHTML=""; fastfoodsEl.innerHTML=""; m.stores.forEach(s=>{ const li=document.createElement("li"); li.textContent=s; storesEl.appendChild(li); }); map.setView([m.location.lat,m.location.lng],15); } // ================= EVENTS ================= searchInput.addEventListener("input",renderMalls); cityFilter.addEventListener("change",renderMalls); closeDetail.addEventListener("click",()=>detailPanel.style.display="none"); // ================= INIT ================= loadMalls();
